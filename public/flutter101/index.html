GraphQLClient createGraphQLClient() {
  final httpLink = HttpLink(
    AppEnv.baseUrl,
    defaultHeaders: {
      "Authorization":
        "Bearer ${AppEnv.bearerToken}",
    },
  );

  return GraphQLClient(
    cache: GraphQLCache(),
    link: httpLink,
  );
}

Chapter 4
the state machine (cubit)

flutter_bloc

our viewmodel.

SearchCubit holds the state and handles user actions.

It executes the data fetch and emits new states. The logic lives here, completely decoupled from the UI.

search/presentation/cubit/search_cubit.dartdart 
class SearchCubit extends Cubit<PageState<List<Book>>> {
  SearchCubit(this._repository) : super(const PageInitial());

  final SearchRepository _repository;

  Future<void> search(String query) async {
    if (query.trim().isEmpty) {
      emit(const PageInitial());
      return;
    }

    emit(const PageLoading());

    final result = await _repository.searchBooks(query);

    switch (result) {
      case Success(:final data):
        emit(PageSuccess(data));
      case Failure(:final exception):
        emit(PageError(exception.message));
    }
  }
}
Chapter 4
reacting to state
widgets/search_results_list.dartdart 
class SearchResultsList extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    // Listens to SearchCubit and rebuilds on new states
    return BlocBuilder<SearchCubit, PageState<List<Book>>>(
      builder: (context, state) {
        
        // Exhaustive pattern matching!
        return switch (state) {
          PageInitial() => const _SearchEmptyState(),
          
          PageLoading() => const PageLoadingIndicator(),
          
          PageSuccess(data: final books) => 
              _SearchSuccessView(books: books),
          
          PageError(message: final msg) => 
              PageErrorView(message: msg),
        };
      },
    );
  }
}

blocbuilder

exhaustive
pattern matching.

Because PageState is sealed, the compiler forces us to handle every state. No more unhandled loading spinners.

offline first

locally.

Network requests are slow and unreliable. We must cache data locally.

We use two distinct tools for this, depending on the complexity of the data we are storing.

🐝

A lightning-fast NoSQL database. We use Hive to store complex, structured objects like user bookshelves (UserBookDto).

A simple key-value store. Perfect for saving primitive, scalar values like the user's daily reading_goal or current_streak.

abstract final class HiveStorage {
  static const bookShelfBox = "bookshelf_box";

  static Future<void> init() async {
    await Hive.initFlutter();

    // Required for complex objects
    Hive.registerAdapters();
    
    await Hive.openBox<UserBookDto>(
      bookShelfBox
    );
  }

  static Box<UserBookDto> getBookShelfBox() {
    return Hive.box<UserBookDto>(bookShelfBox);
  }
}

class PrefsService {
  final asyncPrefs = SharedPreferencesAsync();

  static const _streakKey = "current_streak";
  static const _goalKey = "reading_goal";

  // Async Setters
  Future<void> setCurrentStreak(int n) {
    return asyncPrefs.setInt(_streakKey, n);
  }

  // Async Getters
  Future<int?> getCurrentStreak() {
    return asyncPrefs.getInt(_streakKey);

Chapter 7
consuming storage

repository layer

putting it
together.

How is data actually saved and retrieved?

In our HomeRepository, we inject the services. Hive acts exactly like a Dart Map (using .get and .put), and Prefs provides simple async getters and setters.

home_repository_impl.dartdart 
class HomeRepositoryImpl implements HomeRepository {
  final PrefsService _prefsService; // Injected
  Box<UserBookDto> get _box => HiveStorage.getBookShelfBox();

  // 1. Reading from Hive
  Future<UserBook?> getBookById(String id) async {
    final dto = _box.get(id); // Synchronous read!
    if (dto == null) return null;
    return UserBookMapper.fromDto(dto);
  }

  // 2. Writing to Hive
  Future<void> saveBook(UserBook book) async {
    final dto = UserBookMapper.toDto(book);
    await _box.put(dto.bookId, dto);
    
    await _updateStreak(); // Update user stats
  }

  // 3. Using SharedPreferences
  Future<void> _updateStreak() async {
    final lastRead = await _prefsService.getLastReadDate();
    int streak = await _prefsService.getCurrentStreak() ?? 0;
    
    await _prefsService.setCurrentStreak(streak + 1);
  }
}
